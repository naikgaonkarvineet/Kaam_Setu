9. Supabase PostgreSQL Schema Overview (Technical Reference)
TABLE users ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), phone_number text UNIQUE NOT NULL, name text NOT NULL, role text NOT NULL CHECK (role IN ('worker', 'contractor', 'needs_workers')), location text, preferred_language text DEFAULT 'hindi', created_at timestamptz DEFAULT now() ); TABLE worker_profiles ( id uuid PRIMARY KEY REFERENCES users(id) ON DELETE CASCADE, skills text[] DEFAULT '{}', expertise text, work_experience text, spoken_languages text[] DEFAULT '{}', rating_avg numeric DEFAULT 0 ); TABLE jobs ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), contractor_id uuid REFERENCES users(id), job_type text NOT NULL, location text NOT NULL, wage numeric NOT NULL, working_hours text, num_laborers_required integer DEFAULT 1, start_date date DEFAULT CURRENT_DATE, is_urgent boolean DEFAULT false, status text DEFAULT 'active' CHECK (status IN ('active', 'closed')) ); TABLE applications ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), job_id uuid REFERENCES jobs(id), worker_id uuid REFERENCES users(id), wage numeric NOT NULL, status text DEFAULT 'pending' CHECK (status IN ('pending', 'accepted', 'rejected')) );
